CREATE TABLE EMPLOYEE (
    SSN                 NUMBER          PRIMARY KEY,
    FNAME               VARCHAR2(20)    NOT NULL,
    LNAME               VARCHAR2(20)    NOT NULL,
    BD                  DATE,
    GENDER              char(1)         NOT NULL,
    SUPERVISOR_SSN      NUMBER,
    DNUM                NUMBER
);

CREATE TABLE DEPARTMENT (
    DNUM                    NUMBER          PRIMARY KEY,
    DNAME                   VARCHAR2(20),
    MANAGER_SSN             NUMBER,
    HIRE_DATE               DATE
);

CREATE TABLE DEPARTMENT_LOCATION (
    DLOCATION               VARCHAR2(20),
    DNUM                    NUMBER,
    
    CONSTRAINT PK_DEPARTMENT_LOCATION
    PRIMARY KEY (DNUM, DLOCATION)

);

CREATE TABLE PROJECT (
    PNUM                    NUMBER          PRIMARY KEY,
    PNAME                   VARCHAR2(20),
    PLOCATION               VARCHAR2(20),
    CITY                    VARCHAR2(20),
    DNUM                    NUMBER
);

CREATE TABLE WORKS_ON (
    SSN                     NUMBER,
    PNUM                    NUMBER,
    HOURS                   NUMBER,
    
        CONSTRAINT PK_WORKS_ON
        PRIMARY KEY (SSN, PNUM)
);

CREATE TABLE DEPENDENT (
    SSN                     NUMBER,
    DEPENDENT_NAME          VARCHAR2(20),
    BD                      VARCHAR2(20),
    GENDER                  VARCHAR2(20),
    
            CONSTRAINT PK_DEPENDENT
        PRIMARY KEY (SSN, DEPENDENT_NAME)
);
